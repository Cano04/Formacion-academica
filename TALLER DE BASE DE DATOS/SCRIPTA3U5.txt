CREATE TABLE Usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    correo VARCHAR(100) NOT NULL,
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE Bitacora (
    id INT AUTO_INCREMENT PRIMARY KEY,
    accion VARCHAR(10) NOT NULL,
    id_usuario INT NOT NULL,
    nombre_anterior VARCHAR(50),
    correo_anterior VARCHAR(100),
    nombre_nuevo VARCHAR(50),
    correo_nuevo VARCHAR(100),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id)
);

 create trigger after_usuario_insert
    after insert on Usuarios
    for each row
    begin
    insert into bitacora (accion, id_usuario, nombre_nuevo, correo_nuevo)
    values ('insert', new.id, new.nombre, new.correo);
    end;
