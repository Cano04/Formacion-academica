CREATE FUNCTION calcular_precio_total(costo DECIMAL(10, 2))
RETURNS DECIMAL(10, 2)
DETERMINISTIC
BEGIN
    DECLARE precio_total DECIMAL(10, 2);
    SET precio_total = costo + (costo * 0.16);
    RETURN precio_total;
END
