     (
     ID_CLIENTE           INT NOT NULL DEFAULT 0,
     CLAVE                CHAR(4) NOT NULL,
     NOMBRE               CHAR(255) NOT NULL,
     DOMICILIO            CHAR(255) NOT NULL,
     CIUDAD               CHAR(255),
     ESTADO               CHAR(50), NOT NULL DEFAULT 'SAN LUIS POTOS ',
     ESTATUS              ENUM('ALTA', 'BAJA') NOT NULL DEFAULT 'ALTA',
     KEY (CLAVE),
     KEY (ID_CLIENTE),
     PRIMARY KEY (ID_CLIENTE)
     )ENGINE=INNODB;

CREATE TABLE OrdenTrabajo
     (
     ID_OT                INT NOT NULL DEFAULT 0,
     ID_CLIENTE           INT NOT NULL DEFAULT 0,
     ID_EMPLEADO          INT NOT NULL DEFAULT 0,
     CLAVE_OT             CHAR(4) NOT NULL,
     TIPO_AUTO            CHAR(255) NOT NULL,
     FALLA                CHAR(255),
     FECHA_SOLICITUD      DATETIME,
     PRECIO_FINAL         DOUBLE(15,2) DEFAULT '0.00',
     ESTATUS              ENUM('ABIERTA', 'CERRADA') NOT NULL DEFAULT 'ABIERTA',
     PRIMARY KEY (ID_OT),     
     INDEX(ID_CLIENTE, ID_EMPLEADO),
     FOREIGN KEY (ID_CLIENTE) REFERENCES Clientes(ID_CLIENTE),
     FOREIGN KEY (ID_EMPLEADO) REFERENCES Empleados(ID_EMPLEADO)
     ON UPDATE CASCADE ON DELETE RESTRICT
     )ENGINE=INNODB\g
