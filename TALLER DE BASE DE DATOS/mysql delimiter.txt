CREATE FUNCTION FACTORIAL(X INT) RETURNS INT(11)
    BEGIN
    DECLARE FACTORIAL INT;
    SET FACTORIAL = X;
    IF X<=0 THEN
    RETURN 1;
    END IF;
    BUCLE: LOOP
    SET X = X -1;
    IF X<1 THEN
    LEAVE BUCLE;
    END IF;
    SET FACTORIAL = FACTORIAL *X;
    END LOOP BUCLE;
    RETURN FACTORIAL;
    END
