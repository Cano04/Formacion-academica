USE gente2;
DELIMITER //
CREATE PROCEDURE CONTEOGENERO(OUT CANT_H INT, OUT CANT_M INT)
BEGIN 
SELECT COUNT(*) INTO CANT_H FROM GENTE2.GENTE WHERE SEX='H';
SELECT COUNT(*) INTO CANT_M FROM GENTE2.GENTE WHERE SEX='M';
END
//
DELIMITER ;

CALL CONTEOGENERO(@CH,@CM);
SELECT @CH AS CONTEO_HOMBRES;
SELECT @CM AS CONTEO_MUJERES;


DELIMITER //
CREATE PROCEDURE OBTIENE_ACTIVIDAD(IN OBT_ACT INT(11))
BEGIN
SELECT ACTIVIDAD FROM GENTE2.ACTIVIDAD
JOIN GENTE2.GENTE ON ACTIVIDAD.CCM=GENTE.CCM WHERE GENTE.NUM_PROG=OBT_ACT;
END 
//

DELIMITER ;

CALL OBTIENE_ACTIVIDAD('486647');

DELIMITER //
CREATE PROCEDURE ACTUALIZA_MUN (IN MUN_ACT VARCHAR(225), IN ACT_MUN VARCHAR(255))
BEGIN
UPDATE GENTE2.GENTE SET NOM_MUN=ACT_MUN WHERE NOM_MUN=MUN_ACT;
END
//

DELIMITER ;

CALL ACTUALIZA_MUN('SOLEDAD DE GRACIANO SANCHEZ','SANTA MARIA DEL RIO');

SELECT NOM_MUN FROM GENTE;

DELIMITER //

CREATE PROCEDURE CUENTE_CAP(IN CANT_CAP CHAR(4))
BEGIN
DECLARE CONT_CAP INT;
SELECT COUNT(*) INTO CONT_CAP
FROM GENTE
WHERE CAP=CANT_CAP;
SELECT CONT_CAP AS "TOTAL DE INSCRITOS EN EL CAP";
END
//
DELIMITER ;

CALL CUENTE_CAP('2401');

DELIMITER //

CREATE PROCEDURE NOM_CAP(IN ID_CAP CHAR(4))
BEGIN
DECLARE NOM VARCHAR(250);
SELECT NOMBRE INTO NOM
FROM CENTROS
WHERE CAP =ID_CAP;
SELECT NOM AS "NOMBRE del CAP";
END
//

DELIMITER ;
CALL NOM_CAP('2401');
